cmake_minimum_required(VERSION 3.14)
project(BacktestingEngine CXX)

# Default to an optimized build (about 14x faster than no build type).
# For debugging, configure a separate folder with -DCMAKE_BUILD_TYPE=Debug.
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Your code lives in a library so both the app and the tests can use it
add_library(engine src/engine.cpp src/loader.cpp src/portfolio.cpp src/backtester.cpp src/timestamp.cpp src/aligned.cpp src/multi_backtester.cpp src/metrics.cpp src/zigzag.cpp src/capture.cpp src/param_space.cpp src/optimiser.cpp src/validation.cpp src/dataset.cpp src/report.cpp src/study.cpp src/cli.cpp src/sundays.cpp src/costs.cpp src/baseline_registry.cpp src/rstats.cpp src/control.cpp src/screen.cpp)
target_include_directories(engine PUBLIC include ${CMAKE_BINARY_DIR}/generated)

# Every header in include/baselines/ is a baseline. This writes the list the registry is built from, so adding a
# baseline never needs an edit here or in the code (CMake re-runs by itself when the folder changes).
# The file name must be lowercase letters, digits and underscores; it becomes the name used by --baseline.
file(GLOB BASELINE_HEADERS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/include/baselines/*.hpp)
list(SORT BASELINE_HEADERS)
set(BASELINE_INCLUDES "")
set(BASELINE_VISITS "")
foreach(header ${BASELINE_HEADERS})
  get_filename_component(stem ${header} NAME_WE)
  if(NOT stem MATCHES "^[a-z][a-z0-9_]*$")
    message(FATAL_ERROR "include/baselines/${stem}.hpp: the file name must match [a-z][a-z0-9_]*")
  endif()
  string(APPEND BASELINE_INCLUDES "#include \"baselines/${stem}.hpp\"\n")
  string(APPEND BASELINE_VISITS "    visit(\"${stem}\", std::type_identity<baselines::${stem}::Spec>{});\n")
endforeach()
file(CONFIGURE OUTPUT ${CMAKE_BINARY_DIR}/generated/all_baselines.hpp @ONLY CONTENT
"// Generated by CMake from include/baselines/*.hpp. Do not edit.
#pragma once
#include <type_traits>
@BASELINE_INCLUDES@
template <typename Visit>
void forEachBaseline(Visit&& visit)
{
    (void)visit;
@BASELINE_VISITS@}
")

# runGrid uses std::thread
find_package(Threads REQUIRED)
target_link_libraries(engine PUBLIC Threads::Threads)

# The program you run
add_executable(backtester src/main.cpp)
target_link_libraries(backtester PRIVATE engine)

# Google Test (downloaded automatically during the first configure)
include(FetchContent)
FetchContent_Declare(
  googletest
  URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.zip
)
FetchContent_MakeAvailable(googletest)

enable_testing()
file(GLOB TEST_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/tests/*_test.cpp)
add_executable(engine_tests ${TEST_SOURCES})
target_link_libraries(engine_tests PRIVATE engine GTest::gtest_main)

include(GoogleTest)
gtest_discover_tests(engine_tests)
